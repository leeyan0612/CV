\section*{Research Experience}
\researchentry
  {CAPRA: Multimodal Auditing under Missing Clinical Metadata}
  {ACM MM 2026 Oral}
  {Lead Student Researcher, Beijing University of Posts and Telecommunications}
  {\item Led CAPRA, a semantic-visual multimodal framework that fuses medical images with proxy-axis tokens and calibrated posteriors for missing-metadata subgroup auditing.
   \item Validated fundus, dermoscopy, and chest-radiography tasks under distribution shift, improving worst-group accuracy in 14 of 15 learner--dataset settings.}

\researchentry
  {\paperlink{https://github.com/leeyan0612/CV/raw/refs/heads/main/papers/what-do-medical-fairness-benchmarks-measure.pdf}{Selective-Dependence Auditing for Medical LLM Fairness}}
  {Under Review}
  {Lead Student Researcher, Beijing University of Posts and Telecommunications}
  {\item Led the Selective-Dependence Audit, separating medical judgments, care pathways, support actions, and interface effects to distinguish prohibited dependence from appropriate clinical adaptation.
   \item Built ClinSDA-400 and evaluated six LLMs across 1,075,020 responses in semantic-label, JSON, and MCQ formats, isolating audit displacement caused by option order and answer-symbol mapping.}

\researchentry
  {MedFuse: Anatomical-Prior Fusion for Retinal Lesion Segmentation}
  {FCS 2026}
  {Lead Student Researcher, Beijing University of Posts and Telecommunications}
  {\item Led technical development with an ophthalmologist at Peking University First Hospital, integrating RGB fundus features with retinal vascular priors for diabetic-retinopathy lesion segmentation.
   \item Evaluated a hierarchical dual-encoder fusion pipeline on DDR and IDRiD across five backbones, including annotation-free priors from a multimodal large language model.}
